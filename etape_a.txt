import transmit from '@adonisjs/transmit/services/main'
import type Conversation from '#models/conversation'
import type Message from '#models/message'
import type User from '#models/user'

/**
 * Annonce en temps réel ce qui se passe dans les conversations. Les
 * événements ne contiennent que des identifiants : chaque navigateur
 * demande ensuite au serveur ce qu'il a le droit de voir.
 *
 * La conversation doit avoir ses membres chargés.
 */
export default class RealtimeService {
  /**
   * Un nouveau message a été envoyé.
   */
  messageCreated(conversation: Conversation, message: Message) {
    transmit.broadcast(`conversations/${conversation.id}`, {
      type: 'message.created',
      messageId: message.id,
    })

    this.conversationUpdated(conversation)
  }

  /**
   * Un membre est en train d'écrire dans la conversation.
   */
  userTyping(conversationId: number, user: User) {
    const firstName = user.fullName?.split(' ')[0] ?? user.email.split('@')[0]
    transmit.broadcast(`conversations/${conversationId}`, {
      type: 'user.typing',
      userId: user.id,
      userName: firstName,
    })
  }

  /**
   * Diffuse le changement d'état de présence (en ligne / hors ligne) d'un utilisateur.
   */
  userPresence(userId: number, isOnline: boolean) {
    transmit.broadcast('presence', {
      type: 'user.presence',
      userId,
      isOnline,
    })
  }

  /**
   * Un message a été modifié ou supprimé pour tous.
   */
  messageUpdated(conversation: Conversation, message: Message) {
    transmit.broadcast(`conversations/${conversation.id}`, {
      type: 'message.updated',
      messageId: message.id,
    })

    this.conversationUpdated(conversation)
  }

  /**
   * Prévient chaque membre que la carte de cette conversation, dans sa liste,
   * doit être rafraîchie (aperçu, heure, position).
   */
  conversationUpdated(conversation: Conversation) {
    this.membersChanged(conversation)
  }

  /**
   * Prévient les membres du groupe, ainsi que des utilisateurs en plus
   * (ajoutés ou retirés), de rafraîchir leur liste. Pour un utilisateur
   * retiré, la carte demandée est introuvable : sa liste la retire.
   */
  membersChanged(conversation: Conversation, extraUserIds: number[] = []) {
    const userIds = new Set([
      ...conversation.members.map((member) => member.userId),
      ...extraUserIds,
    ])

    for (const userId of userIds) {
      transmit.broadcast(`users/${userId}`, {
        type: 'conversation.updated',
        conversationId: conversation.id,
      })
    }
  }
}

import type { MultipartFile } from '@adonisjs/core/bodyparser'
import { DateTime } from 'luxon'
import db from '@adonisjs/lucid/services/db'
import Conversation from '#models/conversation'
import Message from '#models/message'
import MessageDeletion from '#models/message_deletion'
import type User from '#models/user'
import { dayKey } from '#services/date_formatter'
import MessageAttachmentService from '#services/message_attachment_service'
import {
  EDIT_WINDOW_MINUTES,
  NOT_HIDDEN_FOR_USER,
  isWithinEditWindow,
} from '#services/message_rules'

/**
 * Levée quand un message ne peut pas (ou plus) être modifié.
 */
export class MessageNotEditableError extends Error {}

/**
 * Règles métier des messages : envoi, lecture, modification et suppression.
 */
export default class MessageService {
  private attachments = new MessageAttachmentService()

  /**
   * Enregistre un message et met à jour la date du dernier message de la
   * conversation (qui sert à trier la liste). Indique aussi si ce message
   * ouvre une nouvelle journée dans la conversation, pour afficher le
   * séparateur de jour.
   */
  async send(user: User, conversation: Conversation, body: string, files: MultipartFile[] = []) {
    return db.transaction(async (trx) => {
      const previous = await Message.query({ client: trx })
        .where('conversation_id', conversation.id)
        .orderBy('id', 'desc')
        .first()

      const trimmed = body.trim()
      const message = await Message.create(
        {
          conversationId: conversation.id,
          senderId: user.id,
          body: trimmed.length > 0 ? trimmed : null,
        },
        { client: trx }
      )

      if (files.length > 0) {
        await this.attachments.attachToMessage(message, files, trx)
      }

      await Conversation.query({ client: trx })
        .where('id', conversation.id)
        .update({ last_message_at: message.createdAt.toJSDate() })

      await message.load('sender')
      await message.load('attachments')

      return {
        message,
        startsNewDay: !previous || dayKey(previous.createdAt) !== dayKey(message.createdAt),
      }
    })
  }

  /**
   * Un message de la conversation, avec son auteur, à condition que
   * l'utilisateur ne l'ait pas masqué. Sinon la recherche échoue et
   * l'application répond « introuvable ».
   */
  async findVisible(user: User, conversationId: number, messageId: number) {
    return Message.query()
      .where('id', messageId)
      .where('conversation_id', conversationId)
      .whereRaw(NOT_HIDDEN_FOR_USER, [user.id])
      .preload('sender')
      .preload('attachments')
      .firstOrFail()
  }

  /**
   * Indique si ce message ouvre une nouvelle journée dans sa conversation.
   */
  async startsNewDay(message: Message) {
    const previous = await Message.query()
      .where('conversation_id', message.conversationId)
      .where('id', '<', message.id)
      .orderBy('id', 'desc')
      .first()

    return !previous || dayKey(previous.createdAt) !== dayKey(message.createdAt)
  }

  /**
   * Messages visibles envoyés après un identifiant donné : sert à rattraper
   * les messages manqués après une coupure de connexion.
   */
  async listAfter(user: User, conversationId: number, afterId: number) {
    const messages = await Message.query()
      .where('conversation_id', conversationId)
      .where('id', '>', afterId)
      .whereRaw(NOT_HIDDEN_FOR_USER, [user.id])
      .preload('sender')
      .preload('attachments')
      .orderBy('id', 'asc')
      .limit(100)

    let previous: Message | null =
      messages.length > 0
        ? await Message.query()
            .where('conversation_id', conversationId)
            .where('id', '<', messages[0].id)
            .orderBy('id', 'desc')
            .first()
        : null

    return messages.map((message) => {
      const startsNewDay = !previous || dayKey(previous.createdAt) !== dayKey(message.createdAt)
      previous = message

      return { message, startsNewDay }
    })
  }

  /**
   * Modifie le texte d'un message de l'utilisateur, pendant le délai autorisé
   * et tant qu'il n'a pas été supprimé pour tous.
   */
  async edit(user: User, conversationId: number, messageId: number, body: string) {
    const message = await this.findOwn(user, conversationId, messageId)

    if (message.deletedAt) {
      throw new MessageNotEditableError('Ce message a été supprimé et ne peut plus être modifié.')
    }
    if (!isWithinEditWindow(message.createdAt)) {
      throw new MessageNotEditableError(
        `Un message ne peut être modifié que pendant ${EDIT_WINDOW_MINUTES} minutes.`
      )
    }

    if (message.body !== body) {
      message.body = body
      message.editedAt = DateTime.now()
      await message.save()
    }

    await message.load('sender')
    await message.load('attachments')

    return message
  }

  /**
   * Supprime un message pour tous les participants. Réservé à son auteur.
   * La suppression est logique : on renseigne la date et on vide le texte.
   */
  async deleteForEveryone(user: User, conversationId: number, messageId: number) {
    const message = await this.findOwn(user, conversationId, messageId)

    if (!message.deletedAt) {
      message.deletedAt = DateTime.now()
      message.body = null
      await message.save()
    }

    return message
  }

  /**
   * Masque un message pour l'utilisateur seulement. Les autres participants
   * le voient toujours. N'importe quel message de la conversation peut être
   * masqué, y compris un message déjà supprimé pour tous.
   */
  async hideForUser(user: User, conversationId: number, messageId: number) {
    const message = await Message.query()
      .where('id', messageId)
      .where('conversation_id', conversationId)
      .firstOrFail()

    await MessageDeletion.firstOrCreate({ messageId: message.id, userId: user.id })

    return message
  }

  /**
   * Message de l'utilisateur dans cette conversation. Dans tout autre cas
   * (autre auteur, autre conversation), la recherche échoue et l'application
   * répond « introuvable ».
   */
  private async findOwn(user: User, conversationId: number, messageId: number) {
    return Message.query()
      .where('id', messageId)
      .where('conversation_id', conversationId)
      .where('sender_id', user.id)
      .preload('attachments')
      .firstOrFail()
  }
}

import db from '@adonisjs/lucid/services/db'
import Conversation from '#models/conversation'
import ConversationMember from '#models/conversation_member'
import Message from '#models/message'
import User from '#models/user'
import { NOT_HIDDEN_FOR_USER } from '#services/message_rules'

export type ConversationWithLastMessage = {
  conversation: Conversation
  lastMessage: Message | null
  unreadCount?: number
}

/**
 * Nombre de messages chargés à l'ouverture d'une conversation.
 */
const HISTORY_LIMIT = 100

/**
 * Règles métier des conversations : liste, accès, historique, création.
 */
export default class ConversationService {
  /**
   * Conversations de l'utilisateur, la plus récente en premier. Un groupe est
   * toujours visible de ses membres ; une conversation privée encore vide
   * n'apparaît que chez celui qui l'a créée.
   */
  async listFor(user: User): Promise<ConversationWithLastMessage[]> {
    const conversations = await Conversation.query()
      .whereHas('members', (members) => {
        members.where('user_id', user.id)
      })
      .where((visible) => {
        visible
          .where('type', 'group')
          .orWhereNotNull('last_message_at')
          .orWhere('created_by', user.id)
      })
      .preload('members', (members) => {
        members.preload('user')
      })
      .orderByRaw('last_message_at DESC NULLS LAST, created_at DESC')

    const conversationIds = conversations.map((item) => item.id)
    const [lastMessages, unreadCounts] = await Promise.all([
      this.lastMessagesOf(conversationIds, user),
      this.unreadCountsFor(conversationIds, user),
    ])

    return conversations.map((conversation) => ({
      conversation,
      lastMessage: lastMessages.get(conversation.id) ?? null,
      unreadCount: unreadCounts.get(conversation.id) ?? 0,
    }))
  }

  /**
   * Renvoie la conversation uniquement si l'utilisateur en est membre.
   * Sinon la recherche échoue et l'application répond « page introuvable »,
   * comme si la conversation n'existait pas.
   */
  async findForMember(user: User, conversationId: number) {
    return Conversation.query()
      .where('id', conversationId)
      .whereHas('members', (members) => {
        members.where('user_id', user.id)
      })
      .preload('members', (members) => {
        members.preload('user')
      })
      .firstOrFail()
  }

  /**
   * Comme findForMember, mais uniquement pour un groupe.
   */
  async findGroupForMember(user: User, conversationId: number) {
    return Conversation.query()
      .where('id', conversationId)
      .where('type', 'group')
      .whereHas('members', (members) => {
        members.where('user_id', user.id)
      })
      .preload('members', (members) => {
        members.preload('user')
      })
      .firstOrFail()
  }

  /**
   * Indique si l'utilisateur est membre de la conversation. Sert aussi à
   * autoriser l'abonnement au canal temps réel de la conversation.
   */
  async isMember(userId: number, conversationId: number) {
    const membership = await ConversationMember.query()
      .where('conversation_id', conversationId)
      .where('user_id', userId)
      .first()

    return membership !== null
  }

  /**
   * Derniers messages d'une conversation vus par l'utilisateur (sans ceux
   * qu'il a masqués), dans l'ordre chronologique, avec leur auteur.
   */
  async history(conversation: Conversation, viewer: User) {
    const messages = await Message.query()
      .where('conversation_id', conversation.id)
      .whereRaw(NOT_HIDDEN_FOR_USER, [viewer.id])
      .preload('sender')
      .preload('attachments')
      .orderBy('id', 'desc')
      .limit(HISTORY_LIMIT)

    return messages.reverse()
  }

  /**
   * Dernier message visible d'une conversation pour l'utilisateur.
   */
  async lastMessageOf(conversation: Conversation, viewer: User) {
    const lastMessages = await this.lastMessagesOf([conversation.id], viewer)

    return lastMessages.get(conversation.id) ?? null
  }

  /**
   * Recherche les messages d'une conversation contenant un terme,
   * sans ceux masqués ou supprimés.
   */
  async search(conversation: Conversation, viewer: User, term: string) {
    const trimmed = term.trim()
    if (!trimmed) {
      return []
    }

    return Message.query()
      .where('conversation_id', conversation.id)
      .whereNull('deleted_at')
      .whereRaw(NOT_HIDDEN_FOR_USER, [viewer.id])
      .whereILike('body', `%${trimmed}%`)
      .preload('sender')
      .orderBy('id', 'desc')
      .limit(50)
  }

  /**
   * Démarre une conversation privée avec un autre utilisateur, ou renvoie
   * celle qui existe déjà entre eux.
   */
  async startPrivate(user: User, otherUserId: number) {
    const other = await User.query()
      .where('id', otherUserId)
      .whereNotNull('email_verified_at')
      .firstOrFail()

    /**
     * Clé identique quel que soit celui qui démarre : « 3-7 » pour les
     * utilisateurs 3 et 7. La base interdit deux conversations avec la
     * même clé.
     */
    const privateKey = [user.id, other.id].sort((a, b) => a - b).join('-')

    const existing = await Conversation.findBy('private_key', privateKey)
    if (existing) {
      await this.ensureMembers(existing, [user.id, other.id])
      return existing
    }

    try {
      return await db.transaction(async (trx) => {
        const conversation = await Conversation.create(
          { type: 'private', privateKey, createdBy: user.id },
          { client: trx }
        )

        await ConversationMember.createMany(
          [
            { conversationId: conversation.id, userId: user.id, role: 'member' },
            { conversationId: conversation.id, userId: other.id, role: 'member' },
          ],
          { client: trx }
        )

        return conversation
      })
    } catch (error) {
      /**
       * Les deux utilisateurs ont démarré la conversation en même temps :
       * la contrainte d'unicité a refusé la seconde, on relit la première.
       */
      const concurrent = await Conversation.findBy('private_key', privateKey)
      if (concurrent) {
        await this.ensureMembers(concurrent, [user.id, other.id])
        return concurrent
      }
      throw error
    }
  }

  /**
   * Quitte une conversation : la discussion disparaît de la liste de
   * l'utilisateur. Pour un groupe, un autre administrateur est promu si besoin.
   */
  async leave(user: User, conversationId: number) {
    const conversation = await this.findForMember(user, conversationId)

    await db.transaction(async (trx) => {
      const membership = await ConversationMember.query({ client: trx })
        .where('conversation_id', conversation.id)
        .where('user_id', user.id)
        .firstOrFail()

      await membership.useTransaction(trx).delete()

      if (conversation.type === 'group') {
        const remaining = await ConversationMember.query({ client: trx })
          .where('conversation_id', conversation.id)
          .orderBy('joined_at', 'asc')

        if (remaining.length > 0 && !remaining.some((member) => member.role === 'admin')) {
          remaining[0].role = 'admin'
          await remaining[0].useTransaction(trx).save()
        }
      }
    })

    return conversation
  }

  private async ensureMembers(conversation: Conversation, userIds: number[]) {
    for (const userId of userIds) {
      await ConversationMember.firstOrCreate(
        { conversationId: conversation.id, userId },
        { role: 'member' }
      )
    }
  }

  /**
   * Dernier message visible de chacune des conversations, en une seule
   * requête, avec son auteur.
   */
  private async lastMessagesOf(conversationIds: number[], viewer: User) {
    const lastMessages = new Map<number, Message>()
    if (conversationIds.length === 0) {
      return lastMessages
    }

    const latestIds = db
      .from('messages')
      .whereIn('conversation_id', conversationIds)
      .whereRaw(NOT_HIDDEN_FOR_USER, [viewer.id])
      .groupBy('conversation_id')
      .max('id')

    const messages = await Message.query()
      .whereIn('id', latestIds)
      .preload('sender')
      .preload('attachments')
    for (const message of messages) {
      lastMessages.set(message.conversationId, message)
    }

    return lastMessages
  }

  /**
   * Nombre de messages non lus pour chacune des conversations.
   */
  async unreadCountsFor(conversationIds: number[], viewer: User): Promise<Map<number, number>> {
    const unreadMap = new Map<number, number>()
    if (conversationIds.length === 0) {
      return unreadMap
    }

    const notHiddenQuery = NOT_HIDDEN_FOR_USER.replace(/messages\./g, 'm.')

    const rows = await db
      .from('messages as m')
      .join('conversation_members as cm', 'cm.conversation_id', 'm.conversation_id')
      .where('cm.user_id', viewer.id)
      .whereIn('m.conversation_id', conversationIds)
      .where('m.sender_id', '!=', viewer.id)
      .whereNull('m.deleted_at')
      .whereRaw('m.id > COALESCE(cm.last_read_message_id, 0)')
      .whereRaw(notHiddenQuery, [viewer.id])
      .groupBy('m.conversation_id')
      .select('m.conversation_id')
      .count('* as total')

    for (const row of rows) {
      unreadMap.set(Number(row.conversation_id), Number(row.total))
    }

    return unreadMap
  }

  /**
   * Marque tous les messages actuels de la conversation comme lus pour cet utilisateur.
   */
  async markAsRead(conversationId: number, userId: number) {
    const latestMessage = await Message.query()
      .where('conversation_id', conversationId)
      .orderBy('id', 'desc')
      .first()

    if (latestMessage) {
      await ConversationMember.query()
        .where('conversation_id', conversationId)
        .where('user_id', userId)
        .where((q) => {
          q.whereNull('last_read_message_id').orWhere('last_read_message_id', '<', latestMessage.id)
        })
        .update({ last_read_message_id: latestMessage.id })
    }
  }
}

import { DateTime } from 'luxon'
import type Conversation from '#models/conversation'
import type Message from '#models/message'
import type User from '#models/user'
import AvatarService from '#services/avatar_service'
import type { ConversationWithLastMessage } from '#services/conversation_service'
import { dayKey, formatDayLabel, formatListDate, formatMessageTime } from '#services/date_formatter'
import { isWithinEditWindow } from '#services/message_rules'

export type ConversationCard = {
  id: number
  title: string
  initials: string
  avatarUrl: string | null
  preview: string
  time: string
  isGroup: boolean
  unreadCount: number
}

export type ConversationHeader = {
  title: string
  initials: string
  avatarUrl: string | null
  isGroup: boolean
  memberCount: number
  otherUserId: number | null
  isOnline: boolean
}

export type MessageAttachmentView = {
  id: number
  originalName: string
  storagePath: string
  mimeType: string
  isImage: boolean
  isVideo: boolean
  sizeLabel: string
}

export type MessageBubble = {
  id: number
  body: string
  isMine: boolean
  isDeleted: boolean
  isEdited: boolean
  canEdit: boolean
  canDeleteForEveryone: boolean
  time: string
  authorName: string | null
  authorColor: number
  attachments: MessageAttachmentView[]
}

export type MessageGroup = {
  label: string
  messages: MessageBubble[]
}

export type GroupMemberView = {
  id: number
  name: string
  initials: string
  avatarUrl: string | null
  isAdmin: boolean
  isMe: boolean
}

/**
 * Prépare les données affichées par les vues des conversations : titres,
 * aperçus, heures, auteurs et regroupement des messages par jour.
 */
export default class ConversationPresenter {
  private avatars = new AvatarService()

  /**
   * Indique si les messages d'une conversation doivent afficher leur auteur.
   */
  showsAuthors(conversation: Conversation) {
    return conversation.type === 'group'
  }

  /**
   * Cartes de la liste des conversations.
   */
  async cards(items: ConversationWithLastMessage[], viewer: User): Promise<ConversationCard[]> {
    return Promise.all(
      items.map(async ({ conversation, lastMessage, unreadCount }) => {
        const identity = await this.identity(conversation, viewer)

        return {
          id: conversation.id,
          ...identity,
          preview: this.previewOf(lastMessage, viewer, this.showsAuthors(conversation)),
          time: lastMessage ? formatListDate(lastMessage.createdAt) : '',
          isGroup: conversation.type === 'group',
          unreadCount: unreadCount ?? 0,
        }
      })
    )
  }

  /**
   * En-tête d'une conversation ouverte.
   */
  async header(conversation: Conversation, viewer: User): Promise<ConversationHeader> {
    const isGroup = conversation.type === 'group'
    const otherMember = !isGroup
      ? conversation.members.find((member) => member.userId !== viewer.id)
      : null

    const otherUser = otherMember?.user
    const isOnline = Boolean(
      otherUser?.lastSeenAt && DateTime.now().diff(otherUser.lastSeenAt, 'minutes').minutes < 2
    )

    return {
      ...(await this.identity(conversation, viewer)),
      isGroup,
      memberCount: conversation.members.length,
      otherUserId: otherUser?.id ?? null,
      isOnline,
    }
  }

  /**
   * Membres d'un groupe : les administrateurs d'abord, puis par ordre
   * alphabétique.
   */
  async members(conversation: Conversation, viewer: User): Promise<GroupMemberView[]> {
    const nameOf = (member: Conversation['members'][number]) =>
      member.user.fullName ?? member.user.email

    const sorted = [...conversation.members].sort((a, b) => {
      const byRole = (a.role === 'admin' ? 0 : 1) - (b.role === 'admin' ? 0 : 1)
      return byRole !== 0 ? byRole : nameOf(a).localeCompare(nameOf(b), 'fr')
    })

    return Promise.all(
      sorted.map(async (member) => ({
        id: member.userId,
        name: nameOf(member),
        initials: member.user.initials,
        avatarUrl: await this.avatars.urlFor(member.user),
        isAdmin: member.role === 'admin',
        isMe: member.userId === viewer.id,
      }))
    )
  }

  /**
   * Une bulle de message, avec les actions que l'utilisateur peut faire
   * dessus. C'est le serveur qui fait respecter ces règles : l'interface ne
   * fait que cacher les boutons non permis. Dans un groupe, la bulle des
   * autres affiche le nom de son auteur (le message doit avoir son auteur
   * chargé).
   */
  bubble(message: Message, viewer: User, showAuthor: boolean): MessageBubble {
    const isDeleted = Boolean(message.deletedAt)
    const isMine = message.senderId === viewer.id

    return {
      id: message.id,
      body: isDeleted ? '' : (message.body ?? ''),
      isMine,
      isDeleted,
      isEdited: Boolean(message.editedAt) && !isDeleted,
      canEdit:
        isMine &&
        !isDeleted &&
        Boolean(message.body?.trim()) &&
        isWithinEditWindow(message.createdAt),
      canDeleteForEveryone: isMine && !isDeleted,
      time: formatMessageTime(message.createdAt),
      authorName: showAuthor && !isMine ? (message.sender?.fullName ?? 'Membre') : null,
      authorColor: message.senderId % 6,
      attachments: isDeleted ? [] : this.attachmentsOf(message),
    }
  }

  /**
   * Données du morceau de HTML renvoyé après une action sur un message : la
   * bulle et, si c'est le premier message du jour, l'intitulé du jour.
   */
  fragment(message: Message, viewer: User, startsNewDay: boolean, showAuthor: boolean) {
    return {
      message: this.bubble(message, viewer, showAuthor),
      dayLabel: startsNewDay ? formatDayLabel(message.createdAt) : null,
    }
  }

  /**
   * Messages regroupés par jour, avec un intitulé pour chaque jour.
   */
  groupMessages(messages: Message[], viewer: User, showAuthor: boolean): MessageGroup[] {
    const groups: MessageGroup[] = []
    let currentDay = ''

    for (const message of messages) {
      const day = dayKey(message.createdAt)

      if (day !== currentDay) {
        groups.push({ label: formatDayLabel(message.createdAt), messages: [] })
        currentDay = day
      }

      groups[groups.length - 1].messages.push(this.bubble(message, viewer, showAuthor))
    }

    return groups
  }

  /**
   * Nom, initiales et photo à afficher pour une conversation : l'autre
   * participant pour une conversation privée, le groupe sinon.
   */
  private async identity(conversation: Conversation, viewer: User) {
    if (conversation.type === 'group') {
      const title = conversation.name ?? 'Groupe'
      return {
        title,
        initials: this.initialsOf(title),
        avatarUrl: await this.avatars.urlForKey(conversation.avatarPath),
      }
    }

    const other = conversation.members.find((member) => member.userId !== viewer.id)?.user
    if (!other) {
      return { title: 'Utilisateur introuvable', initials: '?', avatarUrl: null }
    }

    return {
      title: other.fullName ?? other.email,
      initials: other.initials,
      avatarUrl: await this.avatars.urlFor(other),
    }
  }

  /**
   * Aperçu du dernier message : « Vous : … » pour l'utilisateur, et dans un
   * groupe « Alice : … » pour les autres.
   */
  private previewOf(message: Message | null, viewer: User, isGroup: boolean) {
    if (!message) {
      return 'Aucun message'
    }
    if (message.deletedAt) {
      return 'Message supprimé'
    }

    let text = (message.body ?? '').replace(/\s+/g, ' ').trim()
    if (!text && message.attachments && message.attachments.length > 0) {
      const hasVideo = message.attachments.some((file) => file.mimeType.startsWith('video/'))
      const hasImage = message.attachments.some((file) => file.mimeType.startsWith('image/'))
      if (hasVideo) {
        text = '🎬 Vidéo'
      } else if (hasImage) {
        text = '📷 Photo'
      } else {
        text = '📎 Fichier'
      }
    }

    const clipped = text.length > 80 ? `${text.slice(0, 80)}…` : text

    if (message.senderId === viewer.id) {
      return `Vous : ${clipped}`
    }
    if (isGroup) {
      const firstName = message.sender?.fullName?.split(' ')[0] ?? 'Membre'
      return `${firstName} : ${clipped}`
    }
    return clipped
  }

  private attachmentsOf(message: Message): MessageAttachmentView[] {
    if (!message.attachments?.length) {
      return []
    }

    return message.attachments.map((file) => ({
      id: file.id,
      originalName: file.originalName,
      storagePath: file.storagePath,
      mimeType: file.mimeType,
      isImage: file.mimeType.startsWith('image/'),
      isVideo: file.mimeType.startsWith('video/'),
      sizeLabel: this.formatFileSize(file.sizeBytes),
    }))
  }

  private formatFileSize(bytes: number) {
    if (bytes < 1024) {
      return `${bytes} o`
    }
    if (bytes < 1024 * 1024) {
      return `${Math.round(bytes / 1024)} Ko`
    }
    return `${(bytes / (1024 * 1024)).toFixed(1)} Mo`
  }

  private initialsOf(name: string) {
    return name
      .split(' ')
      .filter(Boolean)
      .slice(0, 2)
      .map((part) => part.charAt(0))
      .join('')
      .toUpperCase()
  }
}

import type Message from '#models/message'
import ConversationPresenter from '#services/conversation_presenter'
import ConversationService from '#services/conversation_service'
import MessageAttachmentService, {
  MessageAttachmentError,
} from '#services/message_attachment_service'
import MessageService, { MessageNotEditableError } from '#services/message_service'
import RealtimeService from '#services/realtime_service'
import {
  catchUpValidator,
  deleteMessageValidator,
  editMessageValidator,
  sendMessageValidator,
} from '#validators/message'
import type { HttpContext } from '@adonisjs/core/http'

/**
 * MessagesController gère les messages d'une conversation : lecture,
 * envoi, modification et suppression. Ces actions répondent par un morceau
 * de HTML (la bulle) quand elles sont appelées par le JavaScript de la page,
 * et par une redirection quand le formulaire est envoyé sans JavaScript.
 * Chaque changement est annoncé en temps réel aux participants.
 */
export default class MessagesController {
  private conversations = new ConversationService()
  private messages = new MessageService()
  private presenter = new ConversationPresenter()
  private realtime = new RealtimeService()

  /**
   * Messages envoyés après un identifiant donné (rattrapage après une
   * coupure de connexion), sous forme de HTML
   */
  async index({ auth, params, request, response, view }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversation = await this.conversations.findForMember(user, Number(params.conversationId))
    const { after } = await request.validateUsing(catchUpValidator)

    const items = await this.messages.listAfter(user, conversation.id, after ?? 0)
    const showAuthor = this.presenter.showsAuthors(conversation)

    const parts = await Promise.all(
      items.map(({ message, startsNewDay }) =>
        view.render('fragments/message', {
          conversationId: conversation.id,
          ...this.presenter.fragment(message, user, startsNewDay, showAuthor),
        })
      )
    )

    return response.type('html').send(parts.join(''))
  }

  /**
   * Une bulle de message pour l'utilisateur connecté, appelée quand un
   * événement temps réel annonce un message nouveau ou modifié. Avec
   * ?bare=1, sans séparateur de jour (pour remplacer une bulle existante).
   */
  async show({ auth, params, request, response, view }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversation = await this.conversations.findForMember(user, Number(params.conversationId))

    const message = await this.messages.findVisible(user, conversation.id, Number(params.messageId))
    const startsNewDay =
      request.input('bare') === '1' ? false : await this.messages.startsNewDay(message)

    const html = await view.render('fragments/message', {
      conversationId: conversation.id,
      ...this.presenter.fragment(
        message,
        user,
        startsNewDay,
        this.presenter.showsAuthors(conversation)
      ),
    })

    return response.type('html').send(html)
  }

  /**
   * Envoie un message dans une conversation dont l'utilisateur est membre
   */
  async store({ auth, params, request, response, session, view }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversation = await this.conversations.findForMember(user, Number(params.conversationId))
    const payload = await request.validateUsing(sendMessageValidator)
    const body = payload.body ?? ''

    let attachments: Awaited<ReturnType<typeof MessageAttachmentService.fromRequest>> = []
    try {
      attachments = MessageAttachmentService.fromRequest(request)
    } catch (error) {
      if (error instanceof MessageAttachmentError) {
        if (this.wantsFragment(request)) {
          return response.status(422).json({ errors: [{ message: error.message }] })
        }

        session.flash('error', error.message)
        return response.redirect().toRoute('conversations.show', { id: conversation.id })
      }
      throw error
    }

    if (!body.trim() && attachments.length === 0) {
      if (this.wantsFragment(request)) {
        return response.status(422).json({
          errors: [{ message: 'Écrivez un message ou joignez un fichier.' }],
        })
      }

      session.flash('error', 'Écrivez un message ou joignez un fichier.')
      return response.redirect().toRoute('conversations.show', { id: conversation.id })
    }

    const { message, startsNewDay } = await this.messages.send(
      user,
      conversation,
      body,
      attachments
    )
    await this.conversations.markAsRead(conversation.id, user.id)
    this.realtime.messageCreated(conversation, message)

    if (!this.wantsFragment(request)) {
      return response.redirect().toRoute('conversations.show', { id: conversation.id })
    }

    const html = await view.render('fragments/message', {
      conversationId: conversation.id,
      ...this.presenter.fragment(
        message,
        user,
        startsNewDay,
        this.presenter.showsAuthors(conversation)
      ),
    })

    return response.status(201).type('html').send(html)
  }

  /**
   * Modifie un message de l'utilisateur et renvoie la bulle mise à jour
   */
  async update({ auth, params, request, response, session, view }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversation = await this.conversations.findForMember(user, Number(params.conversationId))
    const { body } = await request.validateUsing(editMessageValidator)

    let message: Message
    try {
      message = await this.messages.edit(user, conversation.id, Number(params.messageId), body)
    } catch (error) {
      if (error instanceof MessageNotEditableError) {
        if (this.wantsFragment(request)) {
          return response.status(422).json({ errors: [{ message: error.message }] })
        }

        session.flash('error', error.message)
        return response.redirect().toRoute('conversations.show', { id: conversation.id })
      }
      throw error
    }

    this.realtime.messageUpdated(conversation, message)

    if (!this.wantsFragment(request)) {
      return response.redirect().toRoute('conversations.show', { id: conversation.id })
    }

    const html = await view.render('fragments/message', {
      conversationId: conversation.id,
      ...this.presenter.fragment(message, user, false, this.presenter.showsAuthors(conversation)),
    })

    return response.type('html').send(html)
  }

  /**
   * Supprime un message : pour l'utilisateur seul (« me ») ou pour tous
   * (« everyone », réservé à l'auteur)
   */
  async destroy({ auth, params, request, response, view }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversation = await this.conversations.findForMember(user, Number(params.conversationId))
    const { scope } = await request.validateUsing(deleteMessageValidator)
    const messageId = Number(params.messageId)

    if (scope === 'me') {
      await this.messages.hideForUser(user, conversation.id, messageId)

      if (!this.wantsFragment(request)) {
        return response.redirect().toRoute('conversations.show', { id: conversation.id })
      }
      return response.noContent()
    }

    const message = await this.messages.deleteForEveryone(user, conversation.id, messageId)
    this.realtime.messageUpdated(conversation, message)

    if (!this.wantsFragment(request)) {
      return response.redirect().toRoute('conversations.show', { id: conversation.id })
    }

    const html = await view.render('fragments/message', {
      conversationId: conversation.id,
      ...this.presenter.fragment(message, user, false, this.presenter.showsAuthors(conversation)),
    })

    return response.type('html').send(html)
  }

  /**
   * Le JavaScript de la page signale ses appels avec cet en-tête.
   */
  private wantsFragment(request: HttpContext['request']) {
    return request.header('x-chat-fetch') === '1'
  }
}

import ConversationPresenter from '#services/conversation_presenter'
import ConversationService from '#services/conversation_service'
import RealtimeService from '#services/realtime_service'
import { startConversationValidator } from '#validators/conversation'
import type { HttpContext } from '@adonisjs/core/http'

/**
 * ConversationsController gère les conversations de l'utilisateur connecté :
 * liste, ouverture d'une conversation et démarrage d'une conversation privée.
 */
export default class ConversationsController {
  private conversations = new ConversationService()
  private presenter = new ConversationPresenter()
  private realtime = new RealtimeService()

  /**
   * Liste des conversations (page d'accueil de l'utilisateur connecté)
   */
  async index({ auth, view }: HttpContext) {
    const user = auth.getUserOrFail()

    const items = await this.conversations.listFor(user)

    return view.render('pages/conversations/index', {
      conversations: await this.presenter.cards(items, user),
    })
  }

  /**
   * Ouvre une conversation et affiche son historique. Répond « introuvable »
   * si l'utilisateur n'en est pas membre.
   */
  async show({ auth, params, view }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversationId = Number(params.id)

    const conversation = await this.conversations.findForMember(user, conversationId)
    await this.conversations.markAsRead(conversation.id, user.id)

    const [items, messages] = await Promise.all([
      this.conversations.listFor(user),
      this.conversations.history(conversation, user),
    ])

    return view.render('pages/conversations/show', {
      activeId: conversation.id,
      conversations: await this.presenter.cards(items, user),
      header: await this.presenter.header(conversation, user),
      groups: this.presenter.groupMessages(
        messages,
        user,
        this.presenter.showsAuthors(conversation)
      ),
    })
  }

  /**
   * Renvoie la carte d'une conversation (morceau de HTML) pour rafraîchir la
   * liste en temps réel. Si l'utilisateur n'est plus membre, la réponse est
   * « introuvable » et sa liste retire la carte.
   */
  async card({ auth, params, request, view }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversationId = Number(params.id)

    const conversation = await this.conversations.findForMember(user, conversationId)
    const isActive = conversation.id === Number(request.input('active'))

    if (isActive) {
      await this.conversations.markAsRead(conversation.id, user.id)
    }

    const lastMessage = await this.conversations.lastMessageOf(conversation, user)
    const unreadMap = await this.conversations.unreadCountsFor([conversation.id], user)
    const unreadCount = isActive ? 0 : (unreadMap.get(conversation.id) ?? 0)

    const [item] = await this.presenter.cards([{ conversation, lastMessage, unreadCount }], user)

    return view.render('fragments/conversation_card', {
      item,
      active: isActive,
    })
  }

  /**
   * Marque la conversation comme lue par l'utilisateur connecté
   */
  async read({ auth, params, response }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversationId = Number(params.id)

    const conversation = await this.conversations.findForMember(user, conversationId)
    await this.conversations.markAsRead(conversation.id, user.id)

    return response.noContent()
  }

  /**
   * Diffuse le signal « en train d'écrire » aux autres membres de la conversation
   */
  async typing({ auth, params, response }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversationId = Number(params.id)

    const isMember = await this.conversations.isMember(user.id, conversationId)
    if (!isMember) {
      return response.forbidden()
    }

    this.realtime.userTyping(conversationId, user)

    return response.noContent()
  }

  /**
   * Recherche des messages dans une conversation
   */
  async search({ auth, params, request, response }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversationId = Number(params.id)

    const conversation = await this.conversations.findForMember(user, conversationId)
    const term = String(request.input('q', '')).trim()

    const results = await this.conversations.search(conversation, user, term)

    return response.json(
      results.map((msg) => ({
        id: msg.id,
        body: msg.body,
        time: msg.createdAt.toFormat('HH:mm'),
        senderName: msg.sender?.fullName ?? msg.sender?.email,
      }))
    )
  }

  /**
   * Démarre une conversation privée (ou ouvre celle qui existe déjà)
   */
  async store({ auth, request, response, session }: HttpContext) {
    const user = auth.getUserOrFail()
    const { userId } = await request.validateUsing(startConversationValidator)

    if (userId === user.id) {
      session.flash('error', 'Vous ne pouvez pas démarrer une discussion avec vous-même.')
      return response.redirect().toRoute('users.index')
    }

    const conversation = await this.conversations.startPrivate(user, userId)

    return response.redirect().toRoute('conversations.show', { id: conversation.id })
  }

  /**
   * Quitte une conversation : elle disparaît de la liste de l'utilisateur.
   */
  async destroy({ auth, params, response, session }: HttpContext) {
    const user = auth.getUserOrFail()
    const conversationId = Number(params.id)

    const conversation = await this.conversations.findForMember(user, conversationId)
    const notifyUserIds = conversation.members.map((member) => member.userId)

    await this.conversations.leave(user, conversationId)
    this.realtime.membersChanged(conversation, notifyUserIds)

    session.flash('success', 'La discussion a été supprimée de votre liste.')
    return response.redirect().toRoute('dashboard')
  }
}
